cmake_minimum_required(VERSION 3.20)
project(bsc_r72_offline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} /W4 /WX /EHsc /wd4324")
else()
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall -Wextra -Werror -pedantic")
endif()
find_package(Threads REQUIRED)
add_executable(r72_ring_test test_r72.cpp)
target_link_libraries(r72_ring_test PRIVATE Threads::Threads)
enable_testing()
add_test(NAME r72_ring_mpsc COMMAND r72_ring_test)
if(WIN32 AND MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 8)
  enable_language(ASM_MASM)
  add_executable(r72_windows_abi_test windows_context.cpp win64_abi_fixture.asm)
  add_test(NAME r72_win64_fixture COMMAND r72_windows_abi_test)
endif()
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME r72_protocol
           COMMAND ${Python3_EXECUTABLE} -m unittest -v test_protocol)
  set_tests_properties(r72_protocol PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
endif()
