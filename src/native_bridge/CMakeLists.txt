cmake_minimum_required(VERSION 3.16)
project(wh3_bridge_host VERSION 0.5.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(POLICY CMP0091)
 cmake_policy(SET CMP0091 NEW)
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(WH3_ALLOW_UNVALIDATED_NATIVE_ISSUE "Explicit experimental build; NEVER release-approved" OFF)
option(WH3_SANITIZE "ASan/UBSan for the portable test build" OFF)
find_package(Threads REQUIRED)
add_library(wh3_identity_gate STATIC src/identity_gate.cpp)
target_include_directories(wh3_identity_gate PUBLIC include)
target_link_libraries(wh3_identity_gate PUBLIC Threads::Threads)
add_library(wh3_host_core STATIC src/bridge_host.cpp src/packet_tracker.cpp)
target_include_directories(wh3_host_core PUBLIC include)
target_link_libraries(wh3_host_core PUBLIC wh3_identity_gate)
if(WH3_ALLOW_UNVALIDATED_NATIVE_ISSUE)
 target_compile_definitions(wh3_host_core PRIVATE WH3_ALLOW_UNVALIDATED_NATIVE_ISSUE=1)
endif()
add_library(wh3_native_bridge SHARED src/lua_module.cpp)
target_link_libraries(wh3_native_bridge PRIVATE wh3_host_core)
set_target_properties(wh3_native_bridge PROPERTIES PREFIX "")
if(WIN32)
 if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Only Windows x64 is supported")
 endif()
 target_sources(wh3_native_bridge PRIVATE src/platform_windows.cpp)
 target_link_libraries(wh3_native_bridge PRIVATE bcrypt)
else()
 target_sources(wh3_native_bridge PRIVATE src/platform_unavailable.cpp)
endif()
foreach(t wh3_identity_gate wh3_host_core wh3_native_bridge)
 if(MSVC)
  target_compile_options(${t} PRIVATE /W4 /WX /EHsc)
 else()
  target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-misleading-indentation)
 endif()
 if(WH3_SANITIZE AND NOT MSVC)
  target_compile_options(${t} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${t} PUBLIC -fsanitize=address,undefined)
 endif()
endforeach()
enable_testing()
add_executable(test_identity_gate tests/test_identity_gate.cpp)
target_link_libraries(test_identity_gate PRIVATE wh3_identity_gate)
add_test(NAME identity_gate_scenarios COMMAND test_identity_gate)
add_executable(replay_1428 tools/replay_1428.cpp)
target_link_libraries(replay_1428 PRIVATE wh3_identity_gate)
add_test(NAME recorded_1428_fields COMMAND replay_1428 ${CMAKE_CURRENT_SOURCE_DIR}/evidence/runtime_1428.tsv)
add_executable(test_host tests/test_host.cpp src/platform_unavailable.cpp)
target_link_libraries(test_host PRIVATE wh3_host_core)
add_test(NAME native_host_forwarding COMMAND test_host)
add_executable(test_lua_exports tests/test_lua_exports.cpp src/lua_module.cpp)
target_link_libraries(test_lua_exports PRIVATE wh3_host_core)
add_test(NAME lua_export_contract COMMAND test_lua_exports)


add_executable(test_packet_tracker tests/test_packet_tracker.cpp)
target_link_libraries(test_packet_tracker PRIVATE wh3_host_core)
add_test(NAME packet_physical_lifetimes COMMAND test_packet_tracker)
add_executable(test_integrated_host tests/test_integrated_host.cpp src/platform_unavailable.cpp)
target_link_libraries(test_integrated_host PRIVATE wh3_host_core)
add_test(NAME integrated_host_pipeline COMMAND test_integrated_host)

if(WIN32)
 add_executable(backend_smoke tests/backend_smoke.cpp)
 add_test(NAME backend_private_process COMMAND backend_smoke ${CMAKE_CURRENT_SOURCE_DIR}/baseline/minhook.x64.dll)
 add_executable(module_load_smoke tests/module_load_smoke.cpp)
 add_test(NAME module_private_process COMMAND module_load_smoke $<TARGET_FILE:wh3_native_bridge>)
endif()
find_program(WH3_TEXLUA texlua)
if(WH3_TEXLUA)
 add_test(NAME lua_client_fixture COMMAND ${WH3_TEXLUA} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_client.lua ${CMAKE_CURRENT_SOURCE_DIR})
 add_test(NAME baseline_lua_regression COMMAND ${WH3_TEXLUA} ${CMAKE_CURRENT_SOURCE_DIR}/tests/baseline_client/tests/test_client.lua ${CMAKE_CURRENT_SOURCE_DIR}/tests/baseline_client)
endif()
