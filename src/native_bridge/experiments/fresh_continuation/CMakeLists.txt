cmake_minimum_required(VERSION 3.18)
project(bsc_rmb_fresh_continuation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(bsc_patch_core STATIC src/bsc_continuation_patch.cpp)
target_include_directories(bsc_patch_core PUBLIC include)
add_executable(test_native_contract tests/test_native_contract.cpp)
target_link_libraries(test_native_contract PRIVATE bsc_patch_core)
if(UNIX AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64")
 enable_language(ASM)
 add_executable(test_machine_delta tests/test_machine_delta.cpp tests/native_transfer_fixture.S)
 target_link_libraries(test_machine_delta PRIVATE bsc_patch_core)
endif()
enable_testing()
add_test(NAME synthetic_patch_contract COMMAND test_native_contract)
if(TARGET test_machine_delta)
 add_test(NAME real_x64_stolen_span_synthetic_fixture COMMAND test_machine_delta)
endif()
if(WIN32)
 if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "WH3 patch requires x64")
 endif()
 add_library(bsc_fresh_continuation SHARED src/windows_patch.cpp)
 target_link_libraries(bsc_fresh_continuation PRIVATE bsc_patch_core bcrypt)
 target_include_directories(bsc_fresh_continuation PRIVATE include)
 set_target_properties(bsc_fresh_continuation PROPERTIES OUTPUT_NAME "bsc_fresh_continuation_EXPERIMENT_OFF_BY_DEFAULT")
 add_executable(test_windows_private_patch tests/test_windows_private_patch.cpp)
 target_link_libraries(test_windows_private_patch PRIVATE bsc_fresh_continuation)
 add_test(NAME windows_private_patch COMMAND test_windows_private_patch)
endif()
